\section{Proofs} \label{app:proofs}
% ------------------------------------------------------------

Throughout this appendix, $b_m$ denotes the volume of the unit ball in $\R^m$ and $s_{m-1}=mb_m$ the $(m-1)$-dimensional surface measure of the unit sphere $S^{m-1}\subset\R^m$. For any subset $L\subset \R^m$ and any $\delta>0$, let $L^\delta$ denote the set $\{y+v \mid y\in L, \|v\| \leq \delta\}$.

\subsection{Proof of \texorpdfstring{\cref{thm:d_alphan to 0}}{Theorem 1}}

The proof rests on a covering estimate for superlevel sets of a Lipschitz density.

\begin{lemma}\label{lem:covernumber}
  Let $f: \R^m\to [0,\infty)$ be a Lipschitz probability density function with Lipschitz constant $M$, and let $L=f^{-1} [c,\infty)$ be a non-empty superlevel set of $f$ for some $c>0$. For all $\varepsilon\in(0, c/M]$ there exists $\{x_1, \ldots x_l\}\subset L$ such that $L \subseteq \cup_i B(x_i, \varepsilon)$ with $l\leq 2^{m}/(c b_m \varepsilon^m)$.
\end{lemma}
\begin{proof}
  Fix $\varepsilon \in (0, c/M]$ and take a maximal $(\varepsilon/2)$-separated subset $S = \{x_1, x_2, \ldots, x_l\}$ within $L$. This means that $|x_i-x_j|\geq \varepsilon$ for all $i\neq j$ and that no additional point in $L$ can be added to $S$ without breaking this condition. By construction this implies that $L\subseteq \cup_{i} B(x_i, \varepsilon)$, as otherwise any missed location in $L$ could be added as an additional point and the set remain $(\varepsilon/2)$-separated.

  As each $x_i\in L$ we also have $B(x_i, \varepsilon/2)\subset L^{\varepsilon/2}$. Since $f$ is Lipschitz with Lipschitz constant $M$, we have $L^{\varepsilon/2} \subseteq f^{-1}[c-M\varepsilon/2, \infty)$. For $\varepsilon\leq c/M$ we have $B(x_i, \varepsilon/2)\subset f^{-1}[c/2, \infty)$, so
  $$\P(X\in B(x_i, \varepsilon/2))=\int_{B(x_i, \varepsilon/2)}f \,dx \geq \frac{c b_m \varepsilon^m}{2^{m}}.$$
  Each of the $B(x_i, \varepsilon/2)$ is disjoint, so $l\,c b_m \varepsilon^m 2^{-m} \leq \P(X\in \cup_i B(x_i, \varepsilon/2))\leq 1$. We thus get $l\leq 2^{m}/(c b_m \varepsilon^m)$.
\end{proof}

\begin{proof}[Proof of \cref{thm:d_alphan to 0}]
  Choose $c\in(0,1)$ such that for $L=f^{-1}[c, \infty)$ we have $\P(X\notin L)<(1-\gamma)/3$.

  Consider the sequence $r_n:= \big(2^{m+1}/(c b_m n(1-\gamma))\big)^{1/m}$. Note that $r_n \in (0, c/M)$ for large enough $n$; restrict the sequence to $n$ where this holds. By \cref{lem:covernumber} there exists a cover of $L$ by $l_n$ balls $\{B(y_i, r_n)\mid i=1, 2, \ldots l_n\}$, with
  $$
  l_n\leq \frac{2^m}{c b_m r_n^m} =n(1-\gamma)/2 .
  $$

  Any two points lying in the same ball $B(y_i, r_n)$ are at distance at most $2r_n$, so they are joined by an edge in the Rips complex at diameter $2r_n$. This implies that the number of connected components at diameter $2r_n$, when restricted to points inside $L$, is bounded above by $l_n \leq n(1-\gamma)/2 $.

  Let $Y(L,n)$ denote the number of the $n$ sample points that are not in $L$. Then the number of connected components at diameter $2r_n$ is bounded above by $Y(L,n)+n(1-\gamma)/2$. We have $e_{(\omega_n)}\leq 2r_n$ whenever the complex at diameter $2r_n$ has fewer than $n-\omega_n$ connected components, so it suffices to show that
  $$
  Y(L,n)+n(1-\gamma)/2<n-\omega_n
  $$
  for large $n$. As $n-\omega_n>n(1-\gamma)$, it is sufficient to show $Y(L,n)<n(1-\gamma)/2$ for large $n$.

  From the construction of $L$, $Y(L,n)$ is binomial with mean $np$ and variance $np(1-p)$, with $p<(1-\gamma)/3$. Chebyshev's inequality gives
  \begin{align*}
    \P(Y(L,n)>n(1-\gamma)/2) & \leq
    \P(|Y(L,n)-np| >n(1-\gamma)/2-n(1-\gamma)/3) \\
    & = \P(|Y(L,n)-np| >n(1-\gamma)/6)                    \\
    & \leq \frac{\operatorname{Var}(Y)}{(n(1-\gamma)/6)^2}             \\
    & \leq \frac{36np(1-p)}{n^2(1-\gamma)^2}             \\
    & \leq \frac{12}{n(1-\gamma)}.
  \end{align*}

  If there are fewer than $n(1-\gamma)$ connected components at diameter $2r_n$ then $e_{(\omega_n)}\leq 2r_n =: C(f, \gamma) n^{-1/m}$, where $C(f,\gamma)= 2\big(2^{m+1}/(c b_m (1-\gamma))\big)^{1/m}$. Thus $\P\big(e_{(\omega_n)} > C(f, \gamma) n^{-1/m}\big)\leq 12/(n(1-\gamma))$, with $C(f,\gamma)$ dependent on $f$ and $\gamma$ but independent of $n$.
\end{proof}

\subsection{Proof of \texorpdfstring{\cref{cex:fattails}}{Counterexample 6}}

For the remainder of this appendix we consider the distribution on $\R^m$, $m\geq 3$, with density $f(x)=c_m/(1+\|x\|^{m+1})$, where $c_m$ is the appropriate normalising constant. A useful quantity is the number of samples with radius at least $R$: let $A_n^{[R, \infty)}$ be the number of samples lying in $\{x\in \R^m\mid \|x\|\geq R\}.$

We want to show that outside a certain radius $R_n$ all the points are isolated at unit length scale and thus each contribute their own connected component. A sufficient condition is that there be no pair of points within distance $1$ of each other among those of radius at least $R_n-1$. We claim nothing about points in the annulus from radius $R_n-1$ to $R_n$, which could be joined to points of smaller radius.

\begin{lemma}\label{lem:bound edges}
  Fix a sequence of positive numbers $R_n$. Let $S_n$ be a set of $n$ points sampled i.i.d.\ according to density $f$, and let $g(n)$ be any function bounded above by $\sqrt{R_n^{m-1}/\log n}$. Let $Y_n$ be the number of pairs within distance $1$ of each other among the points in $\{x\in \R^m\mid \|x\| > R_n-1\}$. Then
  $$\P\left(Y_n=0 \;\Big|\; A^{[R_n-1, \infty)}_n < \sqrt{\frac{R_n^{m-1}}{\log n}} \text{ and } A^{[R_n, \infty)}>g(n)\right)\to 1$$
  as $n\to \infty$.
\end{lemma}
\begin{proof}
  We first bound the probability that two points of large radius are within distance $1$:
  \begin{align*}
    \P(\|x-y\|\leq 1 \mid \|x\|, \|y\|\geq R-1)&=
    \frac{1}{\P(\|x\|,\|y\|\geq R-1)}\int_{R-1}^\infty \frac{c_m s_{m-1}r^{m-1}}{1+r^{m+1}} \P(y\in B(x,1)\mid |x|=r) \,dr.
  \end{align*}
  If $|x|=r$ with $r$ sufficiently large, then the probability that another sample $y$ lies in $B(x,1)$ is bounded above by $\int_{B(x,1)} c_m/(1+\|z\|^{m+1})\,dz \leq C/r^m$ for some constant $C$ independent of $r$. This implies that there exists a constant $C_1$ independent of $R$ such that
  \begin{align*}
    \P(\|x-y\|\leq 1 \mid \|x\|, \|y\| >R-1)& < (R-1)^2 C_1\int_{R-1}^\infty \frac{1}{r^{m+2}}\,dr,
  \end{align*}
  and hence a constant $C_2$ independent of $R$, for sufficiently large $R$, such that
  $$
  \P(\|x-y\|\leq 1 \mid \|x\|, \|y\| >R-1)\leq \frac{C_2}{R^{m-1}}.
  $$

  For each pair of points the probability of being within distance $1$ is thus bounded above by $C_2/R_n^{m-1}$. Changing the function $g(n)$ does not invalidate this upper bound. Together we get
  \begin{align*}
    \E\left[Y_n \;\Big|\; A^{[R_n-1, \infty)}_n\leq \sqrt{\frac{R_n^{m-1}}{\log n}} \text{ and } A^{[R_n, \infty)}>g(n)\right]
    \leq \left(\frac{R_n^{m-1}}{\log n}\right)\frac{C_2}{2R_n^{m-1}}
    = \frac{C_2}{2\log n}
  \end{align*}
  for some $C_2$ independent of sufficiently large $n$. Since $Y_n$ is a non-negative integer, $\P(Y_n\neq 0\mid\cdot)\leq \E[Y_n\mid\cdot]\to 0$, and the result follows.
\end{proof}

Given the above lemma, our next step is to show that with positive probability there are enough points of radius at least $R_n$, but not too many of radius more than $R_n-1$.

\begin{lemma}\label{lem:bound points}
  Let $S_n$ be a sample of points i.i.d.\ according to $f$, and set $R_n=n^{2/m}/(c_m s_{m-1})$. Then
  $$\P\left( A^{[R_n-1, \infty)}_n < 3n^{1-2/m} \text{ and } A^{[R_n, \infty)}_n > \tfrac{1}{4}n^{1-2/m}\right)>1/3$$
  for $n$ sufficiently large.
\end{lemma}

\begin{proof}
  For each radius $R$, the probability that a sample point has modulus at least $R$ is
  $$
  p_{[R,\infty)}=\int_{R}^\infty \frac{c_m s_{m-1}r^{m-1}}{1+r^{m+1}}\,dr.
  $$
  We have $p_{[R, \infty)}\in \big(c_m s_{m-1}/(2R), c_m s_{m-1}/R\big)$ for $R>1$. Thus $A^{[R_n, \infty)}_n \sim \mathcal{B}(n, p)$, the binomial distribution, for some $p \in (n^{-2/m}/2, n^{-2/m})$. Using Chebyshev's inequality,
  $$\P\left(A_n^{[R_n, \infty)}\in \left(\tfrac{1}{4}n^{1-2/m},2n^{1-2/m}\right)\right)>\tfrac{1}{2}.$$

  Given that a sample point $x$ lies in $B(0,R)$, the probability that $x$ has modulus in $[R-1, R)$ is
  \begin{align*}
    \hat{p}&=\frac{\int_{R-1}^R \frac{c_m s_{m-1}r^{m-1}}{1+r^{m+1}}\,dr }{1-p_{[R, \infty)}}
    <\frac{-\frac{c_m s_{m-1}}{R}+\frac{c_m s_{m-1}}{R-1}}{1-\frac{c_m s_{m-1}}{R}}
    <\frac{2c_m s_{m-1}}{R^2}
  \end{align*}
  for sufficiently large $R$. For $R_n=2n^{2/m}/(c_m s_{m-1})$ we have $\hat{p}_n\leq C_1 n^{-4/m}$ for large $n$, with $C_1$ a constant independent of $n$.

  Let $\hat{A}^{[R_{n}-1, R_n)}_n$ be the number of samples with $\|x\|\in[R_n-1, R_n)$, conditioned on there being $A^{[R_n, \infty)}_n$ samples outside $B(0,R_n)$. Then $\hat{A}^{[R_{n}-1, R_n)}_n \sim \mathcal{B}(n-A^{[R_n, \infty)}_n, \hat{p}_n)$. Coupling $\mathcal{B}(n-A^{[R_n, \infty)}_n, \hat{p}_n)$ to $\mathcal{B}(n, \hat{p}_n)$ implies that $\P(\hat{A}^{[R_n-1, R_n)}_n\leq n^{1-2/m})>2/3$ for large enough $n$. Combining,
  \begin{align*}
    \P&\left( A^{[R_n-1, \infty)}_n < 3n^{1-2/m} \text{ and } A^{[R_n, \infty)}_n > \tfrac{1}{4}n^{1-2/m}\right) \\
    &\geq  \P\left( A^{[R_n-1, \infty)}_n < 3n^{1-2/m} \text{ and } A^{[R_n, \infty)}_n \in \left( \tfrac{1}{4}n^{1-2/m}, 2n^{1-2/m}\right)\right)\\
    & \geq \P\left( \hat{A}^{[R_n-1, R_n)}_n < n^{1-2/m}\;\Big|\;  A^{[R_n, \infty)}_n \in \left( \tfrac{1}{4}n^{1-2/m}, 2n^{1-2/m}\right)\right)\P\left( A^{[R_n, \infty)}_n \in \left( \tfrac{1}{4}n^{1-2/m}, 2n^{1-2/m}\right)\right)\\
    &\geq \tfrac{2}{3}\cdot\tfrac{1}{2}=\tfrac{1}{3}. \qedhere
  \end{align*}
\end{proof}

We can finally combine these to complete the counterexample. The probabilities are far from tight, as we need only show that $e_{(\omega_n)}$ does not converge to $0$.

\begin{proof}[Proof of \cref{cex:fattails}]
  A sufficient condition for $e_{(\omega_n)}\geq 1$ is that the graph joining points within distance $1$ of each other has $\tfrac{1}{4}n^{1-2/m}$ isolated points. One way to obtain this many isolated points is to have $A_n^{[R_n, \infty)}>\tfrac{1}{4}n^{1-2/m}$ and $Y_n=0$. Consider $R_n=n^{2/m}/(c_m s_{m-1})$.

  From \cref{lem:bound points} we have $\P\big( A^{[R_n-1, \infty)}_n < 3n^{1-2/m} \text{ and } A^{[R_n, \infty)}_n > \tfrac{1}{4}n^{1-2/m}\big)>1/3$ for $n$ sufficiently large. Substituting in $R_n$ shows that $3n^{1-2/m}<\sqrt{R_n^{m-1}/\log n}$, so \cref{lem:bound edges} applies and gives $\P\big(Y_n=0\mid A^{[R_n-1, \infty)}_n < 3n^{1-2/m} \text{ and } A^{[R_n, \infty)}_n > \tfrac{1}{4}n^{1-2/m}\big)\to 1$ as $n\to \infty$, hence at least $\tfrac{3}{4}$ for sufficiently large $n$. Combining,
  \begin{align*}
    \P(e_{(\omega_n)}>1)&\geq \P\left(Y_n=0 \text{ and } A^{[R_n-1, \infty)}_n < 3n^{1-2/m} \text{ and } A^{[R_n, \infty)}_n > \tfrac{1}{4}n^{1-2/m}\right)\\
    &\geq \tfrac{1}{3}\cdot\tfrac{3}{4}=\tfrac{1}{4}. \qedhere
  \end{align*}
\end{proof}
